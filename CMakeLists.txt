cmake_minimum_required(VERSION 3.10)
project(PixelLab CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
include_directories(src)
add_executable(phase2_demo src/phase2_demo.cpp)
add_executable(phase3_demo src/phase3_demo.cpp)
add_executable(run_tests tests/tests.cpp)
add_executable(run_viewer_tests tests/viewer_tests.cpp)

# Optional live OpenGL window (needs GLFW)
find_package(glfw3 QUIET)
find_package(OpenGL QUIET)
if(glfw3_FOUND AND OpenGL_FOUND)
  add_executable(window_demo src/window_demo.cpp)
  target_link_libraries(window_demo glfw OpenGL::GL)
endif()
